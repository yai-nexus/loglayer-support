# @yai-loglayer/next v0.8.2 架构优化建议

## 📋 当前架构分析

### 🎯 核心优势
- **统一接口设计**: 通过 `createNextjsLogger` 自动检测环境，提供统一API
- **模块化架构**: 清晰的 server/browser/react/receiver 分层
- **类型安全**: 完整的 TypeScript 支持
- **多入口点**: 支持 `/client` 和 `/server-only` 专用导入

### ⚠️ 当前痛点

#### 1. **接入复杂度过高**
```typescript
// 当前需要的步骤太多
import { createNextjsLogger, LoggerProvider, createNextjsLogReceiver } from '@yai-loglayer/next';

// 1. 创建logger
const logger = createNextjsLogger({ appName: 'my-app' });

// 2. 设置Provider
<LoggerProvider config={{ appName: 'my-app' }}>{children}</LoggerProvider>

// 3. 创建API路由
const logReceiver = createNextjsLogReceiver(serverLogger);
export async function POST(req) { return logReceiver(req); }

// 4. 在组件中使用
const logger = useComponentLogger('ComponentName');
```

#### 2. **异步初始化问题**
- `createNextjsLogger` 返回 Promise，增加使用复杂度
- 服务端 logger 需要 await，打断开发体验
- React hooks 中处理异步状态复杂

#### 3. **配置分散且重复**
- 多处需要配置 `appName`
- server/browser 配置分离，容易不一致
- 环境检测逻辑分散在多个文件

#### 4. **导入路径混乱**
- 主入口、`/client`、`/server-only` 职责不清
- 开发者容易导入错误的模块
- 类型导入和实现导入混合

#### 5. **Client/Server 代码隔离问题** ⚠️ **关键痛点**
- Next.js 严格要求 client/server 代码隔离，不允许相互引用
- 当前架构中 client 和 server logger 混合在同一入口
- 业务开发者容易误导入，导致构建错误
- 缺乏清晰的文件目录区分

#### 6. **业务接入门槛高**
- 需要理解 server/browser 差异
- 手动创建API路由处理客户端日志
- React集成需要多步配置
- 缺乏开箱即用的预设方案

## 🚀 v0.8.2 优化方案

### 核心设计原则
1. **严格代码隔离**: 完全分离 client/server 代码，避免构建错误
2. **目录清晰区分**: 通过文件目录明确区分使用场景
3. **极简接入**: 一行代码开始使用
4. **零配置**: 智能默认配置，按需覆盖
5. **同步优先**: 避免异步初始化
6. **统一配置**: 单一配置源，自动分发
7. **业务友好**: 面向业务场景的API设计

### 3. 统一配置中心

```typescript
// src/shared/config.ts - 共享配置接口（仅类型定义）
export interface NextjsLogConfig {
  appName: string;
  environment?: 'development' | 'production' | 'test';
  level?: 'debug' | 'info' | 'warn' | 'error';
  
  // 统一配置，自动分发到server/browser
  outputs?: {
    console?: boolean;
    http?: {
      enabled?: boolean;
      endpoint?: string;
      batchSize?: number;
    };
    file?: {
      enabled?: boolean;
      path?: string;
    };
    sls?: {
      enabled?: boolean;
      project?: string;
      logstore?: string;
    };
  };
  
  // 高级配置
  advanced?: {
    enablePerformanceTracking?: boolean;
    enableErrorBoundary?: boolean;
    enableDevTools?: boolean;
  };
}

// src/client/config.ts - 客户端配置管理
class ClientConfigManager {
  private static instance: ClientConfigManager;
  private config: NextjsLogConfig | null = null;

  static getInstance() {
    if (!this.instance) {
      this.instance = new ClientConfigManager();
    }
    return this.instance;
  }

  setConfig(config: NextjsLogConfig) {
    this.config = { ...this.getDefaultConfig(), ...config };
  }

  getConfig(): NextjsLogConfig {
    return this.config || this.getDefaultConfig();
  }

  private getDefaultConfig(): NextjsLogConfig {
    return {
      appName: 'nextjs-app',
      environment: 'development',
      level: 'debug',
      outputs: {
        console: true,
        http: {
          enabled: false, // 客户端默认不启用HTTP
          endpoint: '/api/logs',
          batchSize: 10,
        },
      },
    };
  }
}

// src/server/config.ts - 服务端配置管理
class ServerConfigManager {
  private static instance: ServerConfigManager;
  private config: NextjsLogConfig | null = null;

  static getInstance() {
    if (!this.instance) {
      this.instance = new ServerConfigManager();
    }
    return this.instance;
  }

  setConfig(config: NextjsLogConfig) {
    this.config = { ...this.getDefaultConfig(), ...config };
  }

  getConfig(): NextjsLogConfig {
    return this.config || this.getDefaultConfig();
  }

  private getDefaultConfig(): NextjsLogConfig {
    const isDev = process.env.NODE_ENV === 'development';
    return {
      appName: 'nextjs-app',
      environment: process.env.NODE_ENV as any || 'development',
      level: isDev ? 'debug' : 'info',
      outputs: {
        console: isDev,
        file: {
          enabled: !isDev,
          path: './logs',
        },
        sls: {
          enabled: !isDev,
        },
      },
    };
  }
}
```

### 4. 分离的初始化API

```typescript
// src/client/index.ts - 客户端主入口
import { ClientConfigManager } from './config';
import { createBrowserLogger } from './logger';
import { LoggerProvider, useLogger, useComponentLogger } from './react';

/**
 * 客户端日志器初始化
 */
export function initClientLogger(config: NextjsLogConfig) {
  ClientConfigManager.getInstance().setConfig(config);

  const logger = createBrowserLogger(config);

  return {
    // 同步logger方法
    info: (message: string, data?: any) => logger.info(message, data),
    debug: (message: string, data?: any) => logger.debug(message, data),
    warn: (message: string, data?: any) => logger.warn(message, data),
    error: (message: string, data?: any) => logger.error(message, data),

    // 工具方法
    withContext: (context: Record<string, any>) => logger.withContext(context),
    forModule: (moduleName: string) => logger.forModule(moduleName),

    // React集成
    Provider: LoggerProvider,
    useLogger,
    useComponentLogger,
  };
}

// 预设快速启动
export const clientQuickStart = {
  dev: (appName: string) => initClientLogger({ appName, environment: 'development' }),
  prod: (appName: string) => initClientLogger({ appName, environment: 'production' }),
};

// src/server/index.ts - 服务端主入口
import { ServerConfigManager } from './config';
import { createServerLogger } from './logger';
import { createAutoApiRoute } from './receiver';

/**
 * 服务端日志器初始化
 */
export function initServerLogger(config: NextjsLogConfig) {
  ServerConfigManager.getInstance().setConfig(config);

  const logger = createServerLogger(config);

  return {
    // 同步logger方法
    info: (message: string, data?: any) => logger.info(message, data),
    debug: (message: string, data?: any) => logger.debug(message, data),
    warn: (message: string, data?: any) => logger.warn(message, data),
    error: (message: string, data?: any) => logger.error(message, data),

    // 工具方法
    withContext: (context: Record<string, any>) => logger.withContext(context),
    forModule: (moduleName: string) => logger.forModule(moduleName),

    // API路由集成
    createReceiver: () => createAutoApiRoute(logger),
  };
}

// 预设快速启动
export const serverQuickStart = {
  dev: (appName: string) => initServerLogger({ appName, environment: 'development' }),
  prod: (appName: string) => initServerLogger({ appName, environment: 'production' }),
};
```

// src/server/receiver/auto-setup.ts - 服务端自动化设置
export function createAutoApiRoute(logger: any) {
  return {
    // 自动生成的API路由处理器
    POST: async (request: NextRequest) => {
      const receiver = createNextjsLogReceiver(logger, {
        validation: { requireLevel: true },
        processing: { supportBatch: true },
      });

      return receiver(request);
    },
    
    // 支持OPTIONS预检请求
    OPTIONS: async () => {
      return new NextResponse(null, {
        status: 200,
        headers: {
          'Access-Control-Allow-Origin': '*',
          'Access-Control-Allow-Methods': 'POST, OPTIONS',
          'Access-Control-Allow-Headers': 'Content-Type',
        },
      });
    },
  };
}
